\newcommand{\declare}[1]{\newbool{on#1}%

\declare{dynamic}\declare{observation}\declare{action}\declare{reward}\declare{latency}\declare{semantic}

\tikzset{
  every node/.style={text=ink},
  blockbox/.style={draw=neutral, fill=neutralbg, line width=0.9pt, rounded corners=5pt,
                minimum width=5.4cm, align=center, inner sep=6pt},
  shiftbox/.style 2 args={draw=#1, fill=#2, line width=1.0pt, rounded corners=5pt,
                minimum width=4.9cm, minimum height=1.55cm, align=center, inner sep=5pt},
  flow/.style={-{Stealth[length=6pt,width=5pt]}, line width=0.9pt, draw=mute},
  learn/.style={flow, dashed},
  sig/.style={font=\small, text=mute, fill=white, inner sep=1.6pt},
  tagbox/.style 2 args={draw=#1, fill=#2, text=#1, line width=0.7pt, rounded corners=3pt,
                font=\footnotesize\bfseries, inner xsep=4.5pt, inner ysep=2pt, minimum height=13pt},
  cardbox/.style 2 args={draw=#1, fill=#2, line width=1.0pt, rounded corners=5pt,
                text width=7.75cm, align=left, inner sep=7pt},
}

% #1 switch  #2 colour  #3 background  #4 options  #5 node name  #6 position  #7 title  #8 formula
\newcommand{\shiftnode}[8]{%
  \ifbool{on#1}
    {\node[shiftbox={#2}{#3}, #4] (#5) at #6 {{\bfseries\color{#2} #7}\\[1pt] #8};}
    {\node[shiftbox={faint}{white}, #4] (#5) at #6 {{\bfseries\color{faint} #7}};}}

\node[blockbox, minimum height=2.55cm] (env) at (0,0) {};

% ------------------------------------------------------------ the two cross-cutting shifts
% In the overview the cards stand side by side; on a page of its own a card is centred.
\newcommand{\latencycard}[2]{%
  \node[cardbox={lat}{latbg}, anchor=#1] (latency) at #2
    {{\bfseries\color{lat} Latency shift}\; \tikz[baseline=(t.base)]\node[tagbox={lat}{latbg}] (t) {$\delta_o,\ \delta_a,\ \kappa_t$};
     \\[3pt] {\small\color{mute} the timing of sensing, execution and control}\\[5pt]
     $\tilde o_t = \mathcal{I}(\mathcal{H}_t,\, t-\delta_o)$\\[3pt]
     $\tilde a_t = a_{t-\delta_a}$\\[3pt]
     $s_{t+1} \sim P_0^{(\kappa_t)}(\cdot \mid s_t, \tilde a_t)$};}
\newcommand{\semanticcard}[2]{%
  \node[cardbox={sem}{sembg}, anchor=#1] (semantic) at #2
    {{\bfseries\color{sem} Semantic shift}\; \tikz[baseline=(t.base)]\node[tagbox={sem}{sembg}] (t) {$\theta^z$};
     \\[3pt] {\small\color{mute} the scene, the appearance and the task layout}\\[5pt]
     $o_t \sim O_{\theta^z}(\cdot \mid s_t)$\\[3pt]
     $s_{t+1} \sim P_{\theta^z}(\cdot \mid x_t)$\\[3pt]
     $(r_t, c_t) \sim (R_{\theta^z}, C_{\theta^z})(\cdot \mid y_t)$};}

\newcommand{\notes}[1]{%
  \node[font=\small, text=mute, anchor=north] at (0,#1)
    {$x_t = (s_t, \tilde a_t)$ \qquad $y_t = (s_t, \tilde a_t, s_{t+1})$ \qquad
     $\theta$ with a superscript is the parameter of the shift};}
